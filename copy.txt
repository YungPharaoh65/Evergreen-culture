rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 🌿 Garden Forms
    match /gardenForms/{docId} {
      allow read, create, update, delete: if request.auth != null;
    }

    // 👤 Users collection — users can only access their own data
    match /users/{userId} {
      allow read, write, delete: if request.auth != null && request.auth.uid == userId;
    }

    // 👩‍🌾 Gardeners collection
    match /gardeners/{docId} {
      allow read, create, update, delete: if request.auth != null;
    }

    // 🌱 Plants collection
    match /plants/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;
    }

    // 💬 Community Topics
match /communityTopics/{docId} {
  // Anyone can read topics
  allow read: if true;

  // Only signed-in users can create new topics
  allow create: if request.auth != null;

  // Updates — only allow comment modifications
  allow update: if request.auth != null
    && request.resource.data.title == resource.data.title
    && request.resource.data.authorId == resource.data.authorId
    && request.resource.data.createdAt == resource.data.createdAt
    && request.resource.data.keys().hasOnly(["title", "authorId", "createdAt", "comments"]);

  // Only the topic author can delete their own topic
  allow delete: if request.auth != null 
    && (request.auth.uid == resource.data.authorId || request.auth.token.isAdmin == true);
}


    // 🔐 Admin Logins Collection
    match /adminLogins/{docId} {
      allow read, create, update, delete: if request.auth != null;
    }

    // 🗣️ Feedbacks collection
    match /feedbacks/{docId} {
      allow read, create, update, delete: if request.auth != null;
    }
  }
}
