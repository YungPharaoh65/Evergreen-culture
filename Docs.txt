ABOUT PROJECT:

The Evergreen project composes of 3 sections of the entire project 
that is implemented in the creation of the projects:

The client site - what the client sees when they log into their site
The admin dashboard - details of the logged in users as well as their orders and services
The Service Providor - The people advertising their services to the clients who want it done proffesionally   

This site ensures that all parties of the project are satidsfied 
with what it is that they are provided with in terms of service delivery 
and the regard of satisfying both customer and provider

This site is a host for all three areas of business
 B2B
 B2C
 C2C

 it covers up as well the aspects of giving serviceas well as receiving 
 it. The puropse of the Evergreen project is to benefit
 from it by having nature as part of your surrounding, whether
 you are an experienced gardener / farmer looking to explore
 other options of plantation, to a person showing interest

 rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Garden Forms (open access — dev only)
    match /gardenForms/{document=**} {
      allow read, write: if request.auth != null; // Restricted to authenticated users
    }

    // Users collection — users can only access their own data
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Gardeners collection
    match /gardeners/{docId} {
      // Public read access — for dev only; restrict in production
      allow read: if request.auth != null; // Only authenticated users can read

      // Only logged-in users can create entries
      allow create: if request.auth != null;

      // Admin-only update and delete access
      allow update, delete: if request.auth != null && request.auth.uid == get(/databases/$(database)/documents/admins/firstAdmin).id; // Replace with dynamic admin UID
    }

    // Plants collection
    match /plants/{docId} {
      // Only logged-in users can create plant entries
      allow create: if request.auth != null;

      // Read access for everyone, update and delete restricted to admins
      allow read: if true; // Open for everyone (can restrict in production)
      allow update, delete: if request.auth != null && request.auth.token.isAdmin == true;
    }
  }
}
